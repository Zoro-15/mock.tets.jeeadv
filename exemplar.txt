function goToQ(n) {
  // save time for previous question
  if(curQ && curMode === 'attempt') {
    qTimes[String(curQ)] += Math.round((Date.now() - qStartTime) / 1000);
    qVisited[String(curQ)] = true;
    // update per-question timer display
    var ts = qTimes[String(curQ)];
    var el = document.getElementById('qt' + curQ);
    if(el) el.textContent = (ts >= 60 ? Math.floor(ts/60)+':'+pad(ts%60) : '00:'+pad(ts));
  }
  // hide previous card
  var prevCard = document.getElementById('acard' + curQ);
  if(prevCard) prevCard.classList.remove('active-card');

  qStartTime = Date.now();
  curQ = n;
  qVisited[String(n)] = true;

  var q = Q_DATA[n-1];
  curSec = q.si;
  // update sec tabs
  document.querySelectorAll('.stab').forEach(function(b,i){
    b.classList.toggle('active', i === curSec);
  });
  // show current card only
  var card = document.getElementById('acard' + n);
  if(card) {
    card.classList.add('active-card');
    window.scrollTo(0, 0);
  }
  refreshPalette();
}

function secTabClick(si) {

  });
  // go to first q of this section
  if(SEC_QNUMS[si] && SEC_QNUMS[si].length) goToQ(SEC_QNUMS[si][0]);
}

function selectOpt(qn, oi) {
  if(submitted && !reattempt) return;
  if(submitted && reattempt) {
    // reattempt mode: allow re-selection
    answers[String(qn)] = oi;
    renderOpts(qn);
    return;
  }
  // toggle
  if(answers[String(qn)] === oi) {
    answers[String(qn)] = -1;
  } else {
    answers[String(qn)] = oi;
  }
  renderOpts(qn);
  updateQCircle(qn);
  refreshPalette();
}

function renderOpts(qn) {
  var q = Q_DATA[qn-1];
  q.opts.forEach(function(_, oi) {
    var el = document.getElementById('opta_' + qn + '_' + oi);
    if(!el) return;
    el.classList.remove('sel','reveal-correct','reveal-wrong');
    var icon = document.getElementById('oicon_' + qn + '_' + oi);
    if(icon) icon.innerHTML = '';

    if(submitted && !reattempt) {
      if(oi === q.cidx) {
        el.classList.add('reveal-correct');
        if(icon) icon.innerHTML = '<span style="color:#4caf50;font-size:18px">✓</span>';
      } else if(oi === answers[String(qn)] && answers[String(qn)] !== q.cidx) {
        el.classList.add('reveal-wrong');
        if(icon) icon.innerHTML = '<span style="color:#f44336">✗ <span style="color:#aaa;font-size:12px">○</span> <span style="color:#4caf50">↺</span></span>';
      }
    } else if(answers[String(qn)] === oi) {
      el.classList.add('sel');
    }
  });
}

function stateClass(qn) {
  if(submitted) {
    var ans = answers[String(qn)];
    var correct = Q_MAP[String(qn)].correct;
    if(ans === -1) return '';
    return ans === correct ? 'correct' : 'wrong';
  }
  if(marked[String(qn)]) return 'marked';
  if(answers[String(qn)] !== -1) return 'attempted';
  return '';
}

function updateQCircle(qn) {
  var el = document.getElementById('qnc' + qn);
  if(!el) return;
  el.className = 'qnum-circle ' + stateClass(qn);
}

      else if(answers[String(qn)] !== -1) att++;
      else if(!qVisited[String(qn)]) unseen++;
      else unatt++;
    });
    var el = document.getElementById('spstats' + si);
    if(el) el.innerHTML =
      '<div class="sp-stat"><div class="dot" style="background:#f44336"></div><span style="color:#f44336">★ '+mrk+'</span></div>' +
      '<div class="sp-stat"><div class="dot" style="background:#2196f3"></div><span>'+att+'</span></div>' +
      '<div class="sp-stat"><div class="dot" style="background:#bbb"></div><span>'+unatt+'</span></div>' +
      '<div class="sp-stat"><div class="dot" style="border:1.5px solid #ccc;background:#fff"></div><span>'+unseen+'</span></div>';
  });
  // update q circles for all
  Q_DATA.forEach(function(q) { updateQCircle(q.n); });
}

function normCdf(z) {
  // Abramowitz-Stegun approximation of the standard normal CDF.
  var t = 1/(1+0.2316419*Math.abs(z));
  var d = 0.3989423*Math.exp(-z*z/2);
  var p = d*t*(0.3193815+t*(-0.3565638+t*(1.781478+t*(-1.821256+t*1.330274))));
  return z>0 ? 1-p : p;
}

function renderRank(score, maxScore) {
  var card = document.getElementById('ana-rank-card');
  var A = ANALYSIS || {};
  var attempts = A.attempts || 0;
  var maxMarks = A.maxMarks || maxScore || 0;
  var topper   = A.topperMarks || 0;
  var avg      = A.avgMarks;
  // Nothing useful to estimate from → hide the card entirely.
  if(!attempts && (avg===undefined||avg===null) && !topper) { card.style.display='none'; return; }
  // Approximate the score distribution as a normal curve.
  if(avg===undefined||avg===null) avg = topper ? topper*0.45 : maxMarks*0.35;
  var std = (topper && topper>avg) ? (topper-avg)/3 : Math.max(1,(maxMarks||1)*0.15);
  if(std<=0) std = Math.max(1,(maxMarks||1)*0.15);
  var z = (score-avg)/std;
  var pct = normCdf(z)*100;
  if(topper && score>=topper) pct = 99.9;
  pct = Math.max(0.1, Math.min(99.9, pct));
  document.getElementById('ana-percentile').textContent = pct.toFixed(1) + ' %ile';
  if(attempts) {
    var rank = Math.max(1, Math.round((1-pct/100)*attempts));
    if(topper && score>=topper) rank = 1;
    document.getElementById('ana-rank').textContent = '#' + rank.toLocaleString();
    document.getElementById('ana-attempts').textContent = attempts.toLocaleString();
  } else {
    document.getElementById('ana-rank').textContent = '-';
    document.getElementById('ana-attempts').textContent = '-';
  }
  var meta = [];
  if(topper) meta.push('Topper: ' + (topper%1===0?topper:topper.toFixed(1)) + ' / ' + maxMarks);
  if(A.avgMarks!==undefined && A.avgMarks!==null) meta.push('Avg: ' + (A.avgMarks%1===0?A.avgMarks:A.avgMarks.toFixed(1)));
  if(A.cutoff) meta.push('Cutoff: ' + A.cutoff);
  meta.push('Estimated from Testbook attempt data — actual rank may vary.');
  document.getElementById('ana-rank-meta').innerHTML = meta.join(' &nbsp;•&nbsp; ');
  card.style.display = 'block';
}

function showAnalysis() {
  var totalCorrect=0, totalWrong=0, totalSkip=0;
  var totalScore=0, maxScore=0;
  Q_DATA.forEach(function(q) {
    var ans = answers[String(q.n)];
    var pos = Q_MAP[String(q.n)].pos || 1;
    var neg = Q_MAP[String(q.n)].neg || 0;
    maxScore += pos;
    if(ans === -1) { totalSkip++; }
    else if(ans === q.cidx) { totalCorrect++; totalScore += pos; }
    else { totalWrong++; totalScore -= neg; }
  });
  var attempted = totalCorrect + totalWrong;
  var accuracy = attempted > 0 ? Math.round((totalCorrect / attempted) * 100) : 0;

  document.getElementById('ana-score').textContent = totalScore % 1 === 0 ? totalScore : totalScore.toFixed(2);
  document.getElementById('ana-max').textContent = maxScore % 1 === 0 ? maxScore : maxScore.toFixed(2);
  document.getElementById('ana-accuracy').textContent = accuracy + ' %';
  document.getElementById('ana-attempted').textContent = attempted + '/' + TOTAL;
  document.getElementById('ana-correct-count').textContent = totalCorrect;
  document.getElementById('ana-wrong-count').textContent = totalWrong;

  // Estimated rank & percentile (only when Testbook attempt stats exist)
  renderRank(totalScore, maxScore);

  // Question analysis grid
  document.getElementById('ana-grid').innerHTML =
    '<div class="analysis-grid-item ag-correct"><div class="analysis-grid-val">' + totalCorrect + '</div><div class="analysis-grid-label">Correct</div></div>' +
    '<div class="analysis-grid-item ag-wrong"><div class="analysis-grid-val">' + totalWrong + '</div><div class="analysis-grid-label">Incorrect</div></div>' +
    '<div class="analysis-grid-item ag-skip"><div class="analysis-grid-val">' + totalSkip + '</div><div class="analysis-grid-label">Unattempted</div></div>' +
    '<div class="analysis-grid-item ag-total"><div class="analysis-grid-val">' + TOTAL + '</div><div class="analysis-grid-label">Total Questions</div></div>';

  // Section-wise breakdown
  var secHtml = '';
  SEC_NAMES.forEach(function(name, si) {
    var qns = SEC_QNUMS[si];
    var sc=0, wc=0, sk=0, secScore=0, secMax=0;
    qns.forEach(function(qn) {
      var ans = answers[String(qn)];
      var pos = Q_MAP[String(qn)].pos || 1;
      var neg = Q_MAP[String(qn)].neg || 0;
      secMax += pos;
      if(ans === -1) sk++;
      else if(ans === Q_MAP[String(qn)].correct) { sc++; secScore += pos; }
      else { wc++; secScore -= neg; }
    });
    var pct = secMax > 0 ? Math.round(Math.max(0, secScore) / secMax * 100) : 0;
    secHtml += '<div class="analysis-bar-row">' +
      '<span class="analysis-bar-label">' + name + '</span>' +
      '<div class="analysis-bar-track"><div class="analysis-bar-fill" style="width:' + pct + '%;background:' + (pct >= 50 ? '#4caf50' : '#f44336') + '"></div></div>' +
      '<span class="analysis-bar-val">' + (secScore % 1 === 0 ? secScore : secScore.toFixed(1)) + '/' + secMax + '</span>' +
      '</div>';
  });
  document.getElementById('ana-sections').innerHTML = secHtml;

  switchMode('analysis');
}

function startTimer() {
  timerInterval = setInterval(function() {
    if(paused) return;
    timerSecs--;
    if(timerSecs <= 0) { timerSecs = 0; clearInterval(timerInterval); confirmSubmit(); }
    var h = Math.floor(timerSecs/3600);
    var m = Math.floor((timerSecs%3600)/60);
    var s = timerSecs%60;
    var str = pad(h)+':'+pad(m)+':'+pad(s);
    document.getElementById('hdr-timer').textContent = str;
    document.getElementById('modal-timer').textContent = str;
  }, 1000);
}
function pad(n){ return n<10?'0'+n:n; }

  document.getElementById('btn-pause').textContent = paused ? '▶' : '⏸';
}
